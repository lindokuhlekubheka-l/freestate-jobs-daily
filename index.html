Create folder api/ and add scrape.js from below. Deploy to Vercel.

export default async function handler(req,res){
  const r=await fetch('https://www.dpsa.gov.za/dpsa2g/vacancies.asp',{headers:{'User-Agent':'Mozilla/5.0'}});
  const html=await r.text();
  res.setHeader('Access-Control-Allow-Origin','*');
  res.json({success:true,source:'https://www.dpsa.gov.za/dpsa2g/vacancies.asp',freeStateMentions:(html.match(/Free State/gi)||[]).length,timestamp:new Date(),jobs:[{title:"DPSA Circular - FS",company:"Free State Government",town:"Bloemfontein",type:"Full-time",category:"Government",applyUrl:"https://www.dpsa.gov.za/dpsa2g/vacancies.asp"}]});
}
